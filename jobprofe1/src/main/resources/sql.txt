USE bd_spring_batch;

SELECT
    je.JOB_EXECUTION_ID AS ejecucion,
    ji.JOB_NAME AS trabajo,
    je.START_TIME AS inicio,
    je.END_TIME AS fin,
    je.STATUS AS estado,
    TIMESTAMPDIFF(
        SECOND,
        je.START_TIME,
        je.END_TIME
    ) AS duracion_segundos
FROM BATCH_JOB_EXECUTION je
JOIN BATCH_JOB_INSTANCE ji
    ON je.JOB_INSTANCE_ID = ji.JOB_INSTANCE_ID
WHERE ji.JOB_NAME = 'jobGestionClientesSQL'
ORDER BY je.START_TIME DESC
LIMIT 20;
